```sh
python scripts/audit_racing_market_coverage.py --as-of 2026-11-21T10:00:00Z
python scripts/audit_racing_market_coverage.py --as-of 2026-11-21T10:00:00Z --include-pairs
python -m unittest discover -s tests -p 'test*racing_market*.py'
python -m unittest discover -s tests -p 'test_horse_market_archive.py'
```
